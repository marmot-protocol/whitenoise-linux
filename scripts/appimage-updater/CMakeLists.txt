cmake_minimum_required(VERSION 3.20)
project(WhiteNoiseAppImageUpdater LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_LIBAPPIMAGEUPDATE_ONLY ON CACHE BOOL "" FORCE)
set(BUILD_QT_UI OFF CACHE BOOL "" FORCE)
set(LIBAPPIMAGE_DESKTOP_INTEGRATION_ENABLED OFF CACHE BOOL "" FORCE)
set(LIBAPPIMAGE_THUMBNAILER_ENABLED OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_XDGUTILS ON CACHE BOOL "" FORCE)
set(CPR_USE_SYSTEM_CURL ON CACHE BOOL "" FORCE)
add_subdirectory("${WN_UPDATER_SOURCE}" upstream)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WN_GPGME REQUIRED IMPORTED_TARGET gpgme)
add_executable(wn-appimage-update "${CMAKE_CURRENT_LIST_DIR}/../../app/appimage_update.cpp")
target_link_libraries(wn-appimage-update PRIVATE libappimageupdate libappimage PkgConfig::WN_GPGME)
set_target_properties(wn-appimage-update PROPERTIES BUILD_RPATH_USE_ORIGIN ON INSTALL_RPATH "$ORIGIN/appimage-update-lib")
